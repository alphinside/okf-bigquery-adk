### SQL Query

```sql
SELECT
  COUNT(o.order_id) AS total_orders,
  SUM(o.gross_amount) AS total_gmv_idr
FROM `alvin-exploratory-2.marketplace.orders` AS o
JOIN `alvin-exploratory-2.marketplace.payments` AS p
  ON o.order_id = p.order_id
WHERE p.payment_method = 'qris'
  AND p.transaction_status = 'settlement'
  AND o.destination_city IN (
    'Kab. Bekasi',
    'Kab. Bogor',
    'Kab. Tangerang',
    'Kota Bekasi',
    'Kota Bogor',
    'Kota Depok',
    'Kota Jakarta Barat',
    'Kota Jakarta Pusat',
    'Kota Jakarta Selatan',
    'Kota Jakarta Timur',
    'Kota Jakarta Utara',
    'Kota Tangerang',
    'Kota Tangerang Selatan'
  )
  AND o.order_ts >= '2026-07-01'
  AND o.order_ts < '2026-08-01';
```
