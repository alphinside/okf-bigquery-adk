### SQL Query Executed

```sql
SELECT
  SUM(o.gross_amount) AS total_gmv_idr
FROM
  `alvin-exploratory-2.marketplace.orders` AS o
JOIN
  `alvin-exploratory-2.marketplace.payments` AS p
  ON o.order_id = p.order_id
WHERE
  p.transaction_status = 'settlement'
  AND o.order_status = 'completed'
  AND p.payment_method = 'qris'
  AND o.destination_city IN (
    'Kota Jakarta Pusat',
    'Kota Jakarta Utara',
    'Kota Jakarta Barat',
    'Kota Jakarta Selatan',
    'Kota Jakarta Timur',
    'Kota Bogor',
    'Kab. Bogor',
    'Kota Depok',
    'Kota Tangerang',
    'Kota Tangerang Selatan',
    'Kab. Tangerang',
    'Kota Bekasi',
    'Kab. Bekasi'
  )
  AND p.settlement_ts >= TIMESTAMP('2026-07-01 00:00:00+07')
  AND p.settlement_ts < TIMESTAMP('2026-08-01 00:00:00+07');
```
